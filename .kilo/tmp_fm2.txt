240   async createReservation(data, user = {}) {
241     const fId = resourceIdOf(data) || '1';
242     const targetDate =
243       normalizeDate(data.target_date || data.targetDate) || dateOfRange(data.start_time) || todayJst();
244     const startTime = toDateTime(targetDate, data.start_time || data.startTime || '08:50');
245     const endTime = toDateTime(targetDate, data.end_time || data.endTime || '09:40');
246     const title = data.title || data.purpose || '施設利用';
247     const userId = data.user_id || user.id || null;
248     const userName = data.user_name || user.name || '教員';
249     const id = data.id || `res_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`;
250 
251     const overlap = await this.checkOverlap(fId, targetDate, startTime, endTime);
252     if (overlap) {
253       const err = new Error(`指定された時間帯は既存の予約「${overlap.title}」と時間帯が重複しています。`);
254       err.statusCode = 409;
255       throw err;
256     }
257 
258     try {
259       const res = await query(
260         `INSERT INTO reservations (id, resource_id, facility_id, user_id, user_name, title, target_date, start_time, end_time, is_cancelled, created_at)
261          VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9, 0, $10)
262          RETURNING *`,
263         [id, fId, fId, userId, userName, title, targetDate, startTime, endTime, nowIso()]
264       );
265       const created = res.rows[0];
266       if (created) return { ...created, purpose: created.title };
267       return {
268         id,
269         resource_id: fId,
270         facility_id: fId,
271         user_id: userId,
272         user_name: userName,
273         title,
274         target_date: targetDate,
275         start_time: startTime,
276         end_time: endTime,
277         is_cancelled: 0,
278         created_at: nowIso(),
279       };
280     } catch (err) {
281       // 一意制約衝突は 409 として扱う (二重登録防止)
282       if (err && (err.code === '23505' || /duplicate key/i.test(String(err.message)))) {
283         const conflict = new Error('指定された時間帯は既存の予約と時間帯が重複しています。');
284         conflict.statusCode = 409;
285         throw conflict;
286       }
287       throw err;
288     }
289   },

293       `SELECT r.*, COALESCE(u.name, r.user_name) AS user_name
294        FROM reservations r
295        LEFT JOIN users u ON r.user_id = u.id

320   async acquireEditingLock(resource_type, resource_id, user_id, user_name = null) {
321     const nowStr = nowIso();
322     const TTL_MS = 3 * 60 * 1000;
323 
324     const current = await query(
325       'SELECT * FROM editing_locks WHERE resource_type = $1 AND resource_id = $2',
326       [resource_type, resource_id]
327     );
328     const existing = current.rows[0];
329 
330     if (existing && existing.user_id !== user_id) {
331       const last = new Date(existing.last_accessed);
332       if (Date.now() - last.getTime() < TTL_MS) {
333         return null;
334       }
