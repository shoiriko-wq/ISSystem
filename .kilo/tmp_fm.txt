107     const conditions = ['COALESCE(r.is_cancelled, 0) = 0'];
108 
109     if (facilityId) {
110       params.push(String(facilityId));
111       conditions.push(`COALESCE(r.resource_id, r.facility_id) = $${params.length}`);
112     }
113 
114     if (date) {
115       params.push(normalizeDate(date) || date);
116       conditions.push(`(r.target_date = $${params.length} OR substring(r.start_time, 1, 10) = $${params.length})`);
117     }
118 
119     try {
120       const res = await query(
121         `SELECT r.*, COALESCE(u.name, r.user_name) AS user_name, u.avatar_url AS avatar_url,
122                 COALESCE(res.name, r.facility_id) AS facility_name,
123                 COALESCE(r.resource_id, r.facility_id) AS facility_id
124          FROM reservations r
125          LEFT JOIN users u ON r.user_id = u.id
126          LEFT JOIN resources res ON res.id = COALESCE(r.resource_id, r.facility_id)
127          WHERE ${conditions.join(' AND ')}
128          ORDER BY r.target_date, r.start_time`,
129         params
130       );
131       return res.rows.map((r) => ({ ...r, purpose: r.title }));
132     } catch (err) {
133       console.warn('[FacilityModel] getReservationsByDate failed:', err.message);
134       return [];
135     }

157        FROM reservations r
158        LEFT JOIN users u ON r.user_id = u.id
159        WHERE COALESCE(r.resource_id, r.facility_id) = $1
160          AND r.target_date >= $2 AND r.target_date <= $3
161          AND COALESCE(r.is_cancelled, 0) = 0
162        ORDER BY r.target_date, r.start_time`,
163       [String(facilityId), start, end]

170    * @param {string} date - YYYY-MM-DD
171    */
172   async getDailyOverview(date) {
173     const facilities = await this.listFacilities();
174     const reservations = await this.getReservationsByDate(null, date);
175 
176     const byFacility = new Map();
177     for (const r of reservations) {
178       const key = r.resource_id || r.facility_id;
179       if (!byFacility.has(key)) byFacility.set(key, []);
180       byFacility.get(key).push(r);
181     }
182 
183     return facilities.map((facility) => {
184       const list = byFacility.get(facility.id) || [];
185       return {
186         facility,
187         reservations: list,
188         available: list.length === 0,
189       };
190     });
191   },
