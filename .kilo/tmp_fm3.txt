44 /** target_date と時刻表記から文字列比較可能な 'YYYY-MM-DD HH:MM:SS' を組み立てる */
45 function toDateTime(targetDate, time) {
46   const str = String(time || '').trim();
47   if (!str) return null;
48   if (str.includes(' ')) return str; // 既に 'YYYY-MM-DD HH:MM:SS' 形式
49   const date = normalizeDate(targetDate) || todayJst();
50   const hhmm = normalizeTime(str);
51   return hhmm ? `${date} ${hhmm}:00` : `${date} ${str}:00`;
52 }
