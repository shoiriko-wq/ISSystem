        <script>
            const menuGrid = document.getElementById('menuGrid');
            const distributeSection = document.getElementById('distributeSection');
            const submitSection = document.getElementById('submitSection');
            const backToMenuBtn = document.getElementById('backToMenuBtn');
            const headerLogo = document.getElementById('headerLogo');

            const toDistributeCard = document.getElementById('toDistributeCard');
            const toSubmitCard = document.getElementById('toSubmitCard');

            const form = document.getElementById('uploadForm');
            const messageBox = document.getElementById('messageBox');
            const submitModeRadios = document.querySelectorAll('input[name="submitMode"]');
            const fileInputGroup = document.getElementById('fileInputGroup');
            const folderInputGroup = document.getElementById('folderInputGroup');
            const fileInput = document.getElementById('file');
            const folderInput = document.getElementById('folder');
            const folderPreview = document.getElementById('folderPreview');
            const distributeFileList = document.getElementById('distributeFileList');

            function showMenu() {
                menuGrid.style.display = 'grid';
                distributeSection.style.display = 'none';
                submitSection.style.display = 'none';
                backToMenuBtn.style.display = 'none';
                messageBox.style.display = 'none';
            }

            function showDistribute() {
                menuGrid.style.display = 'none';
                distributeSection.style.display = 'block';
                submitSection.style.display = 'none';
                backToMenuBtn.style.display = 'block';
                fetchDistributeFiles();
            }

            function showSubmit() {
                menuGrid.style.display = 'none';
                distributeSection.style.display = 'none';
                submitSection.style.display = 'block';
                backToMenuBtn.style.display = 'block';
            }

            toDistributeCard.addEventListener('click', showDistribute);
            toSubmitCard.addEventListener('click', showSubmit);
            backToMenuBtn.addEventListener('click', showMenu);
            headerLogo.addEventListener('click', showMenu);

            async function fetchDistributeFiles() {
                distributeFileList.innerHTML = '<div class="empty-list">読み込み中...</div>';
                try {
                    const response = await fetch('/api/distributes');
                    const files = await response.json();

                    if (!response.ok || !Array.isArray(files)) {
                        distributeFileList.innerHTML = '<div class="empty-list">配布データの取得に失敗しました。</div>';
                        return;
                    }

                    if (files.length === 0) {
                        distributeFileList.innerHTML = '<div class="empty-list">現在、配布されている資料はありません。</div>';
                        return;
                    }

                    distributeFileList.innerHTML = '';
                    files.forEach(file => {
                        const fileItem = document.createElement('div');
                        fileItem.className = 'file-item';
                        
                        const sizeMB = (file.size / (1024 * 1024)).toFixed(2);
                        const displaySize = sizeMB >= 0.1 ? sizeMB + ' MB' : (file.size / 1024).toFixed(1) + ' KB';

                        fileItem.innerHTML = `
                            <div class="file-info">
                                <span class="file-name">${file.name}</span>
                                <span class="file-meta">サイズ: ${displaySize}</span>
                            </div>
                            <a href="/api/distributes/download/${encodeURIComponent(file.name)}" class="download-btn">
                                <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"/><polyline points="7 10 12 15 17 10"/><line x1="12" y1="15" x2="12" y2="3"/></svg>
                                ダウンロード
                            </a>
                        `;
                        distributeFileList.appendChild(fileItem);
                    });
                } catch (error) {
                    console.error(error);
                    distributeFileList.innerHTML = '<div class="empty-list">通信エラーが発生しました。</div>';
                }
            }

            submitModeRadios.forEach(radio => {
                radio.addEventListener('change', (e) => {
                    if (e.target.value === 'file') {
                        fileInputGroup.style.display = 'block';
                        folderInputGroup.style.display = 'none';
                        fileInput.required = true;
                        folderInput.required = false;
                        folderInput.value = '';
                        folderPreview.style.display = 'none';
                    } else {
                        fileInputGroup.style.display = 'none';
                        folderInputGroup.style.display = 'block';
                        fileInput.required = false;
                        folderInput.required = true;
                        fileInput.value = '';
                    }
                });
            });

            fileInput.required = true;

            folderInput.addEventListener('change', (e) => {
                const files = e.target.files;
                if (files && files.length > 0) {
                    folderPreview.innerHTML = '<strong>アップロード対象ファイル (' + files.length + '件):</strong>';
                    const list = document.createElement('ul');
                    list.style.margin = '4px 0 0 0';
                    list.style.paddingLeft = '20px';
                    
                    const maxShow = Math.min(files.length, 5);
                    for (let i = 0; i < maxShow; i++) {
                        const li = document.createElement('li');
                        li.textContent = files[i].webkitRelativePath || files[i].name;
                        list.appendChild(li);
                    }
                    if (files.length > maxShow) {
                        const li = document.createElement('li');
                        li.textContent = '...ほか ' + (files.length - maxShow) + ' 件';
                        list.appendChild(li);
                    }
                    folderPreview.appendChild(list);
                    folderPreview.style.display = 'block';
                } else {
                    folderPreview.style.display = 'none';
                }
            });

            form.addEventListener('submit', async (e) => {
                e.preventDefault();
                messageBox.style.display = 'none';

                const isFolderMode = document.querySelector('input[name="submitMode"]:checked').value === 'folder';
                const selectedFiles = isFolderMode ? folderInput.files : fileInput.files;

                if (!selectedFiles || selectedFiles.length === 0) {
                    messageBox.className = 'message error';
                    messageBox.textContent = '提出するファイルまたはフォルダを選択してください。';
                    messageBox.style.display = 'block';
                    return;
                }

                const formData = new FormData();
                formData.append('studentName', document.getElementById('studentName').value);
                formData.append('assignmentName', document.getElementById('assignmentName').value);

                for (let i = 0; i < selectedFiles.length; i++) {
                    const relativePath = selectedFiles[i].webkitRelativePath || selectedFiles[i].name;
                    formData.append('paths', relativePath);
                }

                for (let i = 0; i < selectedFiles.length; i++) {
                    formData.append('file', selectedFiles[i]);
                }

                try {
                    messageBox.className = 'message';
                    messageBox.style.backgroundColor = '#eff6ff';
                    messageBox.style.color = '#1e40af';
                    messageBox.style.border = '1px solid #bfdbfe';
                    messageBox.textContent = 'アップロード中... 件数が多い場合や大容量の場合は時間がかかる場合があります。';
                    messageBox.style.display = 'block';

                    const response = await fetch('/api/submit', {
                        method: 'POST',
                        body: formData
                    });

                    const result = await response.json();

                    if (response.ok) {
                        messageBox.className = 'message success';
                        messageBox.textContent = result.message;
                        form.reset();
                        folderPreview.style.display = 'none';
                        document.querySelector('input[name="submitMode"][value="file"]').click();
                    } else {
                        messageBox.className = 'message error';
                        messageBox.textContent = result.error || 'エラーが発生しました。';
                    }
                } catch (error) {
                    messageBox.className = 'message error';
                    messageBox.textContent = 'サーバーとの通信に失敗しました。';
                }
            });
        </script>
